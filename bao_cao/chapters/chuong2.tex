\chapter{PHƯƠNG PHÁP PHÁT HIỆN BẤT THƯỜNG}

Chương này trình bày bài toán cụ thể của đề tài, bộ dữ liệu, giao thức chung, lý do chọn ba phương pháp và cách triển
khai từng phương pháp. Mọi cấu hình nêu ở đây là cấu hình đã chạy để ra kết quả ở Chương 3.

\section{Phát biểu bài toán}

Với mỗi loại sản phẩm, mô hình chỉ được học từ ảnh tốt của loại đó, sau đó phải cho mỗi ảnh kiểm thử hai đầu ra:
điểm bất thường của ảnh $s(x)$ để trả lời ``sản phẩm có lỗi không'', và bản đồ bất thường $A_x(u,v)$ để trả lời
``lỗi nằm ở đâu'' (Hình~\ref{fig:overview}; định nghĩa ở Mục~\ref{sec:anomaly}). Mỗi loại có một mô hình riêng, vì
``bình thường'' của một chai thủy tinh khác hẳn ``bình thường'' của một tấm thảm. Đề tài đánh giá bằng AUROC, chỉ số
không phụ thuộc ngưỡng, nên không chọn ngưỡng quyết định tốt/lỗi.

\section{Tổng quan giải pháp}
\label{sec:tong-quan}

Giải pháp của đề tài là một chuỗi xử lý chung cho cả ba phương pháp; chỉ bước xây dựng mô hình và suy luận là khác
nhau. Với mỗi loại sản phẩm:
\begin{enumerate}
  \item \textbf{Đầu vào:} ảnh \texttt{train/good} để xây dựng mô hình, ảnh test và mặt nạ để đánh giá
        (Mục~\ref{sec:du-lieu}).
  \item \textbf{Tiền xử lý:} đổi ảnh về $256\times256$, chuẩn hóa theo yêu cầu của từng phương pháp, đổi mặt nạ cùng
        kích thước (Mục~\ref{sec:preprocess}).
  \item \textbf{Xây dựng mô hình từ ảnh tốt:} huấn luyện Autoencoder; hoặc trích đặc trưng ResNet-18 và ước lượng
        phân phối Gauss theo vị trí (PaDiM); hoặc trích đặc trưng Wide-ResNet-50-2 và chọn coreset (PatchCore)
        (Mục~\ref{sec:ae} đến~\ref{sec:patchcore}).
  \item \textbf{Suy luận:} với mỗi ảnh test, tính sai số tái tạo, khoảng cách Mahalanobis hoặc khoảng cách tới patch
        gần nhất tại từng vị trí.
  \item \textbf{Bản đồ và điểm bất thường:} đưa các giá trị cục bộ về bản đồ $256\times256$, làm mượt, rồi suy ra
        điểm của ảnh.
  \item \textbf{Đánh giá:} tính image-AUROC từ điểm ảnh, pixel-AUROC từ bản đồ, ghi kèm thời gian và dung lượng
        (Mục~\ref{sec:tieu-chi}).
\end{enumerate}

Mỗi lựa chọn cấu hình trong chuỗi trên đều có lý do (Bảng~\ref{tab:lua-chon}). Nguyên tắc chung là giữ giá trị
mặc định của bài báo gốc và thư viện khi không có lý do để đổi, và không chỉnh bất kỳ giá trị nào theo kết quả trên
tập test.

\section{Bộ dữ liệu MVTec AD}
\label{sec:du-lieu}

cho phát hiện bất thường không giám sát trên ảnh công nghiệp, phát hành theo giấy phép CC BY-NC-SA 4.0 (chỉ dùng
cho mục đích phi thương mại). Đường tải chính thức không truy cập được khi thực hiện đề tài, nên nhóm dùng bản sao
trên Hugging Face và đã đối chiếu tổng số ảnh, số kiểu lỗi với bài báo gốc.

\section{Tiền xử lý và giao thức chung}
\label{sec:preprocess}

Để so sánh công bằng, ba phương pháp dùng chung cách xử lý dữ liệu và cách đánh giá; chỉ phần mô hình bên trong là
khác nhau.
\begin{enumerate}
  \item \textbf{Kích thước ảnh.} Ảnh được đổi về $256\times256$ bằng nội suy song tuyến, không cắt ảnh. Bài báo
        PaDiM và PatchCore cắt vùng giữa $224\times224$; nhóm không cắt để không mất lỗi nằm sát mép ảnh và để bản
        đồ bất thường khớp toàn bộ mặt nạ.
  \item \textbf{Ảnh xám.} Ảnh của grid, screw, zipper được nhân thành 3 kênh để dùng chung mạng với ảnh màu.
  \item \textbf{Chuẩn hóa giá trị điểm ảnh.} PaDiM và PatchCore chuẩn hóa theo trung bình (0,485; 0,456; 0,406) và độ
        lệch chuẩn (0,229; 0,224; 0,225) của ImageNet, đúng phân phối mà backbone đã học.
        Autoencoder giữ giá trị trong $[0,1]$ để so sánh trực tiếp với đầu ra sigmoid.
  \item \textbf{Mặt nạ.} Mặt nạ được đổi kích thước bằng nội suy lân cận gần nhất rồi nhị phân hóa lại (ngưỡng
        0,5); ảnh test tốt được gán mặt nạ toàn 0.
  \item \textbf{Không tăng cường dữ liệu.} Cả ba phương pháp đều không xoay hay lật ảnh. Với một số loại, phép biến
        đổi này tạo ra ảnh trông giống lỗi (metal\_nut lật mặt là một kiểu lỗi) và đưa mẫu ``bình thường'' không có
        thật vào mô hình.
  \item \textbf{Chia dữ liệu.} Giữ nguyên cách chia train/test của bộ dữ liệu. Mô hình chỉ học từ
        \texttt{train/good}; ảnh test chỉ dùng để đánh giá lần cuối. Siêu tham số giữ theo giá trị mặc định của bài
        báo gốc và thư viện, không chỉnh theo kết quả trên tập test.
  \item \textbf{Đánh giá.} Image-AUROC và pixel-AUROC được tính bằng cùng một module (hàm \texttt{roc\_auc\_score}
        của scikit-learn~\cite{scikit-learnRocAucScore}) riêng cho từng loại, rồi lấy trung bình theo loại. Seed
        ngẫu nhiên cố định là 42.
\end{enumerate}

\section{Lựa chọn phương pháp}
\label{sec:chon-pp}

Ba phương pháp được chọn vì ba lý do. Thứ nhất, chúng thuộc ba hướng khác nhau trong Bảng~\ref{tab:approaches}:
tái tạo, mô hình hóa phân phối và bộ nhớ đặc trưng, nên so sánh chúng cho thấy ảnh hưởng của không gian so sánh
(điểm ảnh hay đặc trưng) và cách mô tả ``bình thường''. Thứ hai, chúng tạo thành một chuỗi mà mỗi phương pháp khắc
phục hạn chế chính của phương pháp trước (Hình~\ref{fig:so-sanh}): PaDiM chuyển từ so sánh điểm ảnh sang so sánh đặc
trưng học trước; PatchCore bỏ ràng buộc theo vị trí và giả định phân phối Gauss của PaDiM. Thứ ba, cả ba đều làm
được trong phạm vi môn học: Autoencoder đủ đơn giản để tự xây dựng, PaDiM và PatchCore có sẵn trong thư viện
Anomalib~\cite{akcayAnomalibDeepLearning2022}, và cả ba đều có kết quả công bố trên MVTec AD để đối chiếu.

\begin{figure}[htbp]
  \centering
  \includegraphics[width=\linewidth]{images/duong_so_sanh.png}
  \caption{Ba phương pháp theo không gian biểu diễn, cách mô tả ``bình thường'' và cách tính điểm}
  \label{fig:so-sanh}
\end{figure}

Từ cơ chế của ba phương pháp, nhóm đặt ra ba giả thuyết và kiểm chứng ở Chương 3:
\begin{itemize}
  \item[(G1)] PaDiM và PatchCore vượt Autoencoder ở hầu hết các loại, vì đặc trưng học trước phân biệt lỗi tốt hơn
        sai số từng điểm ảnh.
  \item[(G2)] Khoảng cách giữa PatchCore và PaDiM lớn nhất ở các loại vật thể xoay (screw, hazelnut, metal\_nut), vì
        PaDiM gắn thống kê với vị trí còn PatchCore thì không.
  \item[(G3)] Autoencoder làm tốt hơn ở các loại có nền đơn giản và lỗi lớn, kém ở các loại có lỗi nhỏ hoặc nhiều
        chi tiết khó tái tạo.
\end{itemize}

\section{Autoencoder}
\label{sec:ae}

\subsection{Nguyên lý}

Autoencoder gồm bộ mã hóa $f$ nén ảnh $x$ thành vector tiềm ẩn $z$ và bộ giải mã $g$ dựng lại ảnh từ
$z$~\cite{goodfellowDeepLearning2016}:
\begin{equation}
  z = f(x), \qquad \hat{x} = g(z) = g\bigl(f(x)\bigr) .
  \label{eq:ae-forward}
\end{equation}
Vector $z$ nhỏ hơn nhiều so với ảnh, nên mạng không chép được nguyên ảnh mà phải học những cấu trúc chung của ảnh
tốt. Khi gặp ảnh lỗi, mạng dựng lại theo kiểu ảnh tốt; vùng lỗi bị dựng sai nên có sai số tái tạo lớn
(Hình~\ref{fig:ae}).

\begin{figure}[htbp]
  \centering
  \includegraphics[width=0.95\linewidth]{images/duong_ae.png}
  \caption{Autoencoder và cách tính bản đồ bất thường (đề tài dùng sai số bình phương, không dùng SSIM)}

\subsection{Kiến trúc}

Nhóm tự xây dựng một Autoencoder tích chập bằng PyTorch (Bảng~\ref{tab:ae-arch}). Bộ mã hóa gồm 5 lớp tích chập
bước 2 thu ảnh từ $256\times256$ về $8\times8$, rồi một lớp kết nối đầy đủ nén $256\times8\times8 = 16384$ giá trị
thành vector 256 chiều. Bộ giải mã đối xứng với bộ mã hóa. Mô hình có khoảng 11,9 triệu tham số (45 MB); hai lớp
kết nối đầy đủ ở cổ chai chiếm khoảng 70\%.

Hai lựa chọn thiết kế quyết định khả năng phát hiện lỗi:
\begin{itemize}
  \item \textbf{Không có kết nối tắt giữa bộ mã hóa và bộ giải mã} như kiến trúc U-Net. Kết nối tắt cho phép chép
        thẳng chi tiết từ đầu vào sang đầu ra, khi đó cả vùng lỗi cũng được dựng lại đúng và tín hiệu bất thường
        mất đi.
  \item \textbf{Cổ chai là một vector}, không phải bản đồ không gian. Cổ chai quá rộng thì mạng dựng lại được cả
        lỗi; quá hẹp thì ảnh tốt cũng bị dựng mờ và gây báo động giả. Vector 256 chiều là điểm cân bằng nhóm chọn
        và giữ cố định cho cả 15 loại.
\end{itemize}

\subsection{Huấn luyện}

Hàm mất mát là sai số bình phương trung bình trên mọi điểm ảnh và kênh màu:
\begin{equation}
  L(x,\hat{x}) = \frac{1}{HWC}\sum_{u=1}^{H}\sum_{v=1}^{W}\sum_{c=1}^{C}
  \bigl[x(u,v,c) - \hat{x}(u,v,c)\bigr]^{2} .
  \label{eq:ae-loss}
\end{equation}
Mỗi loại được huấn luyện 100 epoch bằng Adam, tốc độ học $10^{-3}$, weight decay $10^{-5}$, lô 16 ảnh. Hàm mất mát
dựa trên SSIM~\cite{bergmannImprovingUnsupervisedDefect2019}, so sánh cấu trúc cục bộ thay vì từng điểm ảnh, là một
lựa chọn khác được báo cáo tốt hơn ở một số loại; đề tài chưa thử và ghi nhận ở phần Hạn chế.

\subsection{Bản đồ và điểm bất thường}

Bản đồ bất thường là sai số bình phương tại từng điểm ảnh, lấy trung bình theo kênh:
\begin{equation}
  A(u,v) = \frac{1}{C}\sum_{c=1}^{C}\bigl[x(u,v,c) - \hat{x}(u,v,c)\bigr]^{2} .
  \label{eq:ae-map}
\end{equation}
Bản đồ được làm mượt bằng bộ lọc Gauss với $\sigma = 4$ để giảm nhiễu từng điểm ảnh, rồi lấy giá trị lớn nhất làm
điểm của ảnh.

\subsection{Ưu điểm và hạn chế}

Autoencoder đơn giản, dễ giải thích (xem trực tiếp được ảnh dựng lại), mô hình nhỏ và không phụ thuộc mạng học
trước. Hạn chế nằm ở chính không gian so sánh: ảnh dựng lại thường mờ ở cạnh và kết cấu chi tiết, nên sai số xuất
hiện cả ở vùng bình thường; mạng đủ mạnh có thể dựng lại được cả lỗi; và sai khác từng điểm ảnh nhạy với ánh sáng
hay dịch chuyển nhỏ nhưng kém nhạy với lỗi mang tính ngữ nghĩa. Ngoài ra, mạng phải huấn luyện từ đầu và kết quả
phụ thuộc nhiều vào siêu tham số.

\section{PaDiM}
\label{sec:padim}

\subsection{Nguyên lý}

PaDiM~\cite{defardPaDiMPatchDistribution2021} chia ảnh thành lưới các vùng nhỏ (patch). Mỗi patch được mô tả bằng
vector đặc trưng lấy từ mạng tích chập học trước. Phương pháp giả định rằng, trên ảnh tốt, các vector tại cùng một
vị trí tuân theo một phân phối Gauss nhiều chiều; patch nào lệch xa phân phối của vị trí đó là bất thường
(Hình~\ref{fig:padim}). PaDiM không lan truyền ngược: phần ``huấn luyện'' chỉ là một lượt duyệt ảnh tốt để ước
lượng thống kê.

\begin{figure}[htbp]
  \centering
  \includegraphics[width=\linewidth]{images/duong_padim.png}
  \caption{Quy trình PaDiM (với ảnh $256\times256$ của đề tài, lưới là $64\times64$ và $d = 100$)}
  \label{fig:padim}
\end{figure}

\subsection{Trích xuất đặc trưng}

Đặc trưng lấy từ ba tầng \texttt{layer1}, \texttt{layer2}, \texttt{layer3} của ResNet-18 (Mục 1.3). Tầng nông giữ
chi tiết cục bộ, tầng sâu hơn mang ngữ cảnh rộng; kết hợp cả ba để bắt được cả lỗi nhỏ lẫn lỗi lớn. Bản đồ của
\texttt{layer2} và \texttt{layer3} được phóng về kích thước của \texttt{layer1} rồi ghép theo kênh, cho lưới
$64\times64$ vị trí, mỗi vị trí một vector $64+128+256 = 448$ chiều.

Kích thước ma trận hiệp phương sai tăng theo bình phương số chiều, nên PaDiM giữ lại $d = 100$ chiều chọn ngẫu nhiên
(cố định theo seed, dùng chung cho huấn luyện và kiểm thử). Bài báo gốc cho thấy chọn ngẫu nhiên cho kết quả tương
đương phân tích thành phần chính nhưng nhanh hơn nhiều~\cite{defardPaDiMPatchDistribution2021}.

\subsection{Ước lượng phân phối và tính điểm}

Gọi $z_k(p)$ là vector đặc trưng của ảnh tốt thứ $k$ tại vị trí $p$ và $N$ là số ảnh tốt:
\begin{align}
  \mu_p &= \frac{1}{N}\sum_{k=1}^{N} z_k(p), \label{eq:padim-mean}\\
  \Sigma_p &= \frac{1}{N-1}\sum_{k=1}^{N}\bigl[z_k(p)-\mu_p\bigr]\bigl[z_k(p)-\mu_p\bigr]^{\top} + \varepsilon I .
  \label{eq:padim-cov}
\end{align}
Thành phần $\varepsilon I$ với $\varepsilon = 0{,}01$ bảo đảm $\Sigma_p$ khả nghịch; điều này cần thiết vì mỗi loại
chỉ có 60 đến 391 ảnh tốt, có loại ít hơn số chiều $d$. Với ảnh kiểm thử, điểm tại $p$ là khoảng cách Mahalanobis
từ $z(p)$ tới phân phối bình thường:
\begin{equation}
  d_p = \sqrt{\bigl[z(p)-\mu_p\bigr]^{\top}\,\Sigma_p^{-1}\,\bigl[z(p)-\mu_p\bigr]} .
  \label{eq:padim-mahalanobis}
\end{equation}
Khác khoảng cách Euclid, khoảng cách này tính đến độ biến thiên và tương quan giữa các chiều: lệch trên một chiều
vốn ổn định bị phạt nặng hơn lệch cùng độ lớn trên một chiều vốn dao động nhiều. Bản đồ $64\times64$ các giá trị
$d_p$ được phóng về $256\times256$ và làm mượt Gauss $\sigma = 4$; điểm ảnh là giá trị lớn nhất của bản đồ.

\subsection{Triển khai và bộ nhớ}

Nhóm dùng lớp PaDiM của Anomalib 2.7.0~\cite{akcayAnomalibDeepLearning2022}, giữ nguyên trọng số ResNet-18 học
trước. Mô hình của mỗi loại gồm 4096 vector trung bình và 4096 ma trận nghịch đảo $100\times100$, khoảng 158 MB ở
dạng số thực 32 bit. Đây cũng là lý do dùng ResNet-18 thay vì Wide-ResNet-50-2 như bài báo gốc: với
Wide-ResNet-50-2 và $d = 550$, cùng lưới $64\times64$ cần $4096 \times 550^2$ phần tử, khoảng 4,6 GB cho mỗi loại.

\subsection{Ưu điểm và hạn chế}

PaDiM không cần huấn luyện mạng, chỉ một lượt duyệt dữ liệu nên xây dựng mô hình nhanh, và tận dụng được đặc trưng
học trước nên khoanh vùng lỗi tốt. Hạn chế chính là giả định ảnh đã căn chỉnh: khi vật thể xoay hay dịch chuyển,
phân phối tại một vị trí trộn nhiều bộ phận khác nhau và kém phân biệt hơn. Ngoài ra, bộ nhớ tăng theo bình phương
số chiều, và một phân phối Gauss duy nhất không mô tả đủ khi tại cùng vị trí có nhiều kiểu bình thường khác nhau.

\section{PatchCore}
\label{sec:patchcore}

\subsection{Nguyên lý}

PatchCore~\cite{rothTotalRecallIndustrial2022} coi ``bình thường'' là những gì đã xuất hiện trong ảnh tốt: một patch
kiểm thử là bình thường nếu nó giống ít nhất một patch của ảnh tốt, ở bất kỳ vị trí nào trong bất kỳ ảnh nào
(Hình~\ref{fig:patchcore}). So với PaDiM, phương pháp có hai khác biệt: không phụ thuộc vị trí, và không giả định
dạng phân phối mà dùng trực tiếp các vector mẫu.

\begin{figure}[htbp]
  \centering
  \includegraphics[width=\linewidth]{images/duong_patchcore.png}
  \caption{Quy trình PatchCore (đề tài giữ coreset 10\%)}
  \label{fig:patchcore}
\end{figure}

\subsection{Đặc trưng patch có ngữ cảnh lân cận}

Đặc trưng lấy từ \texttt{layer2} (512 kênh, lưới $32\times32$) và \texttt{layer3} (1024 kênh, $16\times16$) của
Wide-ResNet-50-2. \texttt{layer1} quá gần mức điểm ảnh, \texttt{layer4} quá thiên về phân loại ImageNet. Tại mỗi vị
trí, đặc trưng được lấy trung bình trên lân cận $3\times3$ để mỗi vector mang thông tin vùng xung quanh và bền hơn
với dịch chuyển nhỏ. Bản đồ của \texttt{layer3} được phóng về $32\times32$ rồi ghép với \texttt{layer2}, nên mỗi
ảnh cho 1024 vector 1536 chiều.

\subsection{Bộ nhớ và chọn coreset}

Gom vector của mọi patch trong mọi ảnh tốt được bộ nhớ đầy đủ $\mathcal{M}$: một loại vài trăm ảnh có vài trăm
nghìn vector, tìm láng giềng gần nhất trên đó rất chậm. PatchCore giữ lại một tập con $\mathcal{M}_C$ sao cho mọi
vector trong $\mathcal{M}$ đều có một đại diện gần nó trong $\mathcal{M}_C$:
\begin{equation}
  \mathcal{M}_C^{*} = \arg\min_{\mathcal{M}_C \subset \mathcal{M}} \; \max_{m \in \mathcal{M}} \; \min_{n \in \mathcal{M}_C} \lVert m - n\rVert_2 .
  \label{eq:coreset}
\end{equation}
Tìm nghiệm đúng của bài toán này quá tốn kém, nên PatchCore giải gần đúng bằng thuật toán tham lam
k-center~\cite{senerActiveLearningConvolutional2018}:
\begin{enumerate}
  \item Chiếu ngẫu nhiên mọi vector trong $\mathcal{M}$ xuống ít chiều hơn, để tính khoảng cách nhanh hơn mà vẫn giữ
        gần đúng khoảng cách giữa chúng.
  \item Khởi tạo $\mathcal{M}_C$ bằng một vector chọn ngẫu nhiên.
  \item Với mỗi vector trong $\mathcal{M}$, cập nhật khoảng cách tới vector gần nhất của $\mathcal{M}_C$.
  \item Thêm vào $\mathcal{M}_C$ vector có khoảng cách đó lớn nhất, tức vector đang được đại diện kém nhất.
  \item Lặp bước 3 và 4 đến khi $\mathcal{M}_C$ đủ kích thước mong muốn.
\end{enumerate}
Khác lấy mẫu ngẫu nhiên, cách chọn này giữ được cả những kiểu patch bình thường hiếm gặp, nhờ đó giảm báo động giả.
Bài báo gốc cho thấy giữ 1\% đến 10\% số vector vẫn cho kết quả gần như bộ nhớ đầy
đủ~\cite{rothTotalRecallIndustrial2022}.

Đề tài giữ 10\% số vector. Ví dụ hazelnut có 391 ảnh tốt, tức khoảng 400 nghìn vector; coreset còn khoảng 40 nghìn
vector, chiếm khoảng 235 MB. Vì mỗi bước chọn phụ thuộc bước trước, thời gian chọn coreset tăng nhanh theo số ảnh.

\subsection{Tính điểm}

Điểm của một patch $q$ trong ảnh kiểm thử là khoảng cách tới patch bình thường gần nhất trong coreset:
\begin{equation}
  a(q) = \min_{m \in \mathcal{M}_C}\,\lVert q - m\rVert_{2} .
  \label{eq:patchcore-score}
\end{equation}
Bản đồ $32\times32$ các giá trị $a(q)$ được phóng về $256\times256$ và làm mượt Gauss $\sigma = 4$. Điểm của ảnh
không chỉ là patch có $a(q)$ lớn nhất. Gọi $q^{*}$ là patch đó, $m^{*}$ là láng giềng gần nhất của nó và
$\mathcal{N}_b(m^{*})$ là $b$ láng giềng gần nhất của $m^{*}$ trong coreset; điểm được nhân thêm một trọng số:
\begin{equation}
  s = \left(1 - \frac{\exp\lVert q^{*} - m^{*}\rVert_2}{\sum_{m \in \mathcal{N}_b(m^{*})} \exp\lVert q^{*} - m\rVert_2}\right) a(q^{*}) .
  \label{eq:patchcore-reweight}
\end{equation}
Trọng số này giảm điểm khi $m^{*}$ nằm trong vùng dày đặc của bộ nhớ, tức $q^{*}$ gần nhiều mẫu bình thường chứ
không chỉ một mẫu. Đề tài dùng $b = 9$, đúng như cài đặt của Anomalib 2.7.0.

\subsection{Ưu điểm và hạn chế}

PatchCore không phụ thuộc vị trí nên chịu được vật thể xoay hay lệch, không cần huấn luyện mạng và không giả định
dạng phân phối. Đổi lại, mỗi patch kiểm thử phải tìm láng giềng gần nhất trong coreset nên suy luận chậm hơn PaDiM,
và bước chọn coreset tốn thời gian khi nhiều ảnh. Việc bỏ thông tin vị trí cũng có mặt trái: từng patch của một vật
thể thiếu linh kiện hay lắp sai chỗ vẫn có thể giống một patch bình thường nào đó, nên PatchCore dễ bỏ sót lỗi
logic. Cũng như PaDiM, chất lượng phụ thuộc vào đặc trưng học từ ImageNet.

\section{Cài đặt và chạy thực nghiệm}
\label{sec:cai-dat}

\subsection{Tổ chức mã nguồn}

Mã nguồn viết bằng Python, tổ chức theo nguyên tắc ``cố định cách đánh giá, tự do cách triển khai''. Hai phần dùng
chung cho cả ba phương pháp là module đọc dữ liệu (tiền xử lý như Mục~\ref{sec:preprocess}) và module đánh giá
(tính AUROC, vẽ hình, ghi kết quả). Mỗi phương pháp là một module riêng, chỉ cần nhận ảnh từ module dữ liệu và trả về
bản đồ cùng điểm bất thường đúng kích thước $256\times256$. Nhờ vậy, đổi cách cài đặt một phương pháp không làm thay
đổi cách đánh giá, và kết quả của ba phương pháp so sánh được với nhau. Autoencoder do nhóm tự viết bằng PyTorch;
PaDiM và PatchCore dùng trực tiếp lớp mô hình của Anomalib 2.7.0, nhưng nhận ảnh từ module dữ liệu của nhóm thay vì
bộ nạp dữ liệu của thư viện, để bảo đảm cùng tiền xử lý với Autoencoder. Cấu trúc thư mục chi tiết nằm ở Phụ lục.

\subsection{Quy trình chạy một loại sản phẩm}

Với mỗi phương pháp và mỗi loại, chương trình thực hiện:
\begin{enumerate}
  \item Đặt lại seed ngẫu nhiên bằng 42, để kết quả của mỗi loại không phụ thuộc thứ tự chạy.
  \item Đọc ảnh \texttt{train/good} và toàn bộ tập test của loại đó qua module dữ liệu.
  \item Xây dựng mô hình từ ảnh tốt: huấn luyện mạng (Autoencoder), ước lượng phân phối (PaDiM) hoặc trích đặc trưng
        và chọn coreset (PatchCore).
  \item Suy luận trên toàn bộ tập test để lấy bản đồ bất thường $256\times256$ và điểm của từng ảnh.
  \item Tính image-AUROC và pixel-AUROC bằng module đánh giá chung.
  \item Ghi một dòng vào tệp kết quả chung (loại, phương pháp, hai chỉ số AUROC, thời gian xây dựng, thời gian suy
        luận, dung lượng mô hình, cấu hình) và lưu ảnh minh họa cho từng kiểu lỗi. Với Autoencoder, trọng số mạng của
        từng loại cũng được lưu lại.
\end{enumerate}

\subsection{Cách chạy trên Colab}

Thực nghiệm đầy đủ chạy trên Google Colab; cấu hình phần cứng, phiên bản thư viện và các thiết lập khác được tổng
hợp ở Mục~\ref{sec:thiet-lap}. Dữ liệu và kết quả đặt trên Google Drive, nên khi phiên Colab bị ngắt thì không mất
gì. Mỗi loại chạy xong được ghi ngay vào tệp kết quả; khi chạy lại,
chương trình bỏ qua các loại đã có kết quả với cùng cấu hình và chạy tiếp từ loại còn thiếu. Trước khi chạy đủ 15
loại, mỗi phương pháp được chạy thử trên loại bottle ở máy cá nhân (GPU RTX 3050, 4 GB) để kiểm tra toàn bộ quy
trình.

\subsection{Các vấn đề kỹ thuật đã xử lý}

\begin{itemize}
  \item \textbf{PaDiM lỗi trên GPU Colab.} Phép nghịch đảo đồng thời 4096 ma trận hiệp phương sai trên GPU Tesla T4 gây
        lỗi truy cập bộ nhớ trong thư viện CUDA đi kèm PyTorch trên Colab. Nhóm tính trung bình, hiệp phương sai và
        nghịch đảo trên CPU rồi mới chuyển kết quả lên GPU để suy luận. Kết quả không đổi: loại bottle cho cùng
        image-AUROC 0,9976 và pixel-AUROC 0,9776 trước và sau khi sửa.
  \item \textbf{PatchCore trên GPU 4 GB.} Khi chạy thử ở máy cá nhân, bộ nhớ đặc trưng đầy đủ không vừa GPU. Nhóm giữ
        toàn bộ đặc trưng trên RAM và chỉ đưa bản chiếu ngẫu nhiên lên GPU khi chọn coreset; thuật toán không đổi.
  \item \textbf{Autoencoder đọc lại ảnh mỗi epoch.} Cách nạp dữ liệu thông thường giải nén lại từng ảnh PNG
        độ phân giải cao từ Google Drive ở mỗi epoch, tức 100 lần cho mỗi ảnh. Nhóm đọc và thu nhỏ ảnh huấn luyện
        một lần, giữ sẵn trên GPU (tối đa khoảng 300 MB với 391 ảnh), rồi xáo và chia lô ngay trên GPU. Cấu hình huấn
        luyện giữ nguyên; 100 epoch trên bottle còn khoảng 85 giây trên RTX 3050.
\end{itemize}

\subsection{Đo thời gian và vẽ hình minh họa}

Với Autoencoder, thời gian xây dựng chỉ tính vòng huấn luyện, sau khi đã đọc và chuyển toàn bộ ảnh huấn luyện lên thiết bị. Với PaDiM và PatchCore, thời gian này tính từ lúc duyệt ảnh huấn luyện đến khi hoàn tất trích đặc trưng, ước lượng phân phối hoặc chọn coreset. Thời gian suy luận là tổng thời gian chạy mô hình trên tập test chia
cho số ảnh test. Trước và sau mỗi lần đo, chương trình chờ GPU làm xong mọi phép tính để số đo không bị sai do GPU
chạy bất đồng bộ. Với PaDiM và PatchCore, thời gian suy luận chỉ gồm lượt chạy mô hình; với Autoencoder, số đo còn
gồm cả thời gian đọc ảnh test. Khác biệt này được lưu ý khi so sánh ở Chương 3.

Hình minh họa gồm ảnh gốc, mặt nạ lỗi và bản đồ bất thường của ảnh đầu tiên trong mỗi kiểu lỗi. Thang màu của bản đồ
cố định cho mỗi phương pháp và mỗi loại, lấy từ phân vị 1\% và 99,5\% của mọi bản đồ test, nên các ảnh cùng loại so
sánh được với nhau mà không bị chuẩn hóa riêng từng ảnh.
